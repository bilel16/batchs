PATCH post-snapshot — mettre à jour engagementDetailleJson après création
Deux appels séquentiels — création du snapshot puis enrichissement
Complexité : Moyenne Changements BDD : ZÉRO Backend : 1 endpoint Frontend : 15 lignes
ℹ️
Principe : La création du snapshot se fait normalement via POST /api/avis. La réponse contient le snapshotId. Le frontend fait immédiatement un second appel PATCH /api/canevas/{id}/snapshots/{snapshotId}/engagement-detaille avec les données réelles. Le backend met à jour uniquement le champ engagementDetailleJson du snapshot existant.

  • Retry possible si le premier appel réussit mais pas le second
  • Peut enrichir des snapshots existants rétrospectivement

  • 2 appels HTTP au lieu d'un → légère latence

// ✅ Nouveau endpoint — met à jour uniquement engagementDetailleJson
@PatchMapping("/{id}/snapshots/{snapshotId}/engagement")
public ResponseEntity<Void> updateEngagementDetaille(
        @PathVariable Long id,
        @PathVariable Long snapshotId,
        @RequestBody Map<String, String> body) {

    String engJson = body.get("engagementDetailleJson");
    if (engJson == null || engJson.isBlank()) return ResponseEntity.badRequest().build();

    // Charger le snapshot existant, mettre à jour uniquement ce champ
    snapshotRepo.findById(snapshotId).ifPresent(snap -> {
        snap.setEngagementDetailleJson(engJson);
        snapshotRepo.save(snap);
    });

    return ResponseEntity.noContent().build();
}

private final SnapshotRepository legacySnapshotRepo;   // ← injecter en plus

// Dans buildEngagementDetaille() — remplacer par :
private String buildEngagementDetailleFromLegacy(Long numSeqPers) {
    // 1. Trouver le dernier snapshot legacy pour ce numSeqPers
    return legacySnapshotRepo.findByNumSeqpers(numSeqPers)
            .stream()
            .max(Comparator.comparing(Snapshot::getDateCreation))  // le plus récent
            .map(snap -> {
                // 2. Le JSON legacy a la structure :
                // { relation:{...}, associe:{...}, premierResponsable:{...}, groupe:{...}, vendeur:{...} }
                // On le restructure en EngagementDetailleDto
                try {
                    ObjectMapper mapper = new ObjectMapper();
                    JsonNode root = mapper.readTree(snap.getEngagements());
                    // Extraire et restructurer :
                    EngagementDetailleDto dto = EngagementDetailleDto.builder()
                        .actionnaires(root.has("associe") ?
                            List.of(mapper.treeToValue(root.get("associe"), EngagementPartieDto.class)) : List.of())
                        .premierResponsable(root.has("premierResponsable") ?
                            mapper.treeToValue(root.get("premierResponsable"), EngagementPartieDto.class) : null)
                        .groupe(root.has("groupe") ?
                            List.of(mapper.treeToValue(root.get("groupe"), EngagementPartieDto.class)) : List.of())
                        .vendeurs(root.has("vendeur") ?
                            List.of(mapper.treeToValue(root.get("vendeur"), EngagementPartieDto.class)) : List.of())
                        .build();
                    return toJson(dto);
                } catch (Exception e) {
                    log.warn("Erreur lecture snapshot legacy", e);
                    return null;
                }
            })
            .orElse(null);
}

package net.kachout.saeb.service;

import org.springframework.stereotype.Service;
import java.util.concurrent.ConcurrentHashMap;

@Service
public class EngagementStagingService {

    // clé = canevasId, valeur = JSON de l'engagementDetaille
    private final ConcurrentHashMap<Long, String> cache = new ConcurrentHashMap<>();

    // Stocker le JSON pour un canvas
    public void store(Long canevasId, String engagementDetailleJson) {
        cache.put(canevasId, engagementDetailleJson);
    }

    // Récupérer et supprimer (consommation unique)
    public String consumeAndClear(Long canevasId) {
        return cache.remove(canevasId);
    }

    // Récupérer sans supprimer
    public String peek(Long canevasId) {
        return cache.get(canevasId);
    }
}
