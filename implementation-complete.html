package net.kachout.saeb.service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import net.kachout.saeb.entities.CanevasWorkflowLog;
import net.kachout.saeb.repository.CanevasWorkflowLogRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.*;

@Service
@RequiredArgsConstructor
@Slf4j
public class WorkflowLogService {

    private final CanevasWorkflowLogRepository workflowRepo;

    // ── Version complète avec structureMere ──────────────────────────────────
    @Transactional
    public CanevasWorkflowLog log(
            Long canevasId,
            String fromStructure, String toStructure,
            String action,
            Integer snapshotVersion, Long snapshotId,
            String comments, String performedBy,
            String structureMere) {

        CanevasWorkflowLog entry = CanevasWorkflowLog.builder()
                .canevasId(canevasId)
                .fromStructure(fromStructure)
                .toStructure(toStructure)
                .action(action)
                .snapshotVersion(snapshotVersion)
                .snapshotId(snapshotId)
                .comments(comments)
                .performedBy(performedBy)
                .structureMere(structureMere)
                .build();

        CanevasWorkflowLog saved = workflowRepo.save(entry);
        log.info("WorkflowLog: canevas={} {} -[{}]-> {}", canevasId, fromStructure, action, toStructure);
        return saved;
    }

    // ── Surcharge rétrocompatible (pour CanevasSnapshotController) ───────────
    @Transactional
    public CanevasWorkflowLog log(
            Long canevasId,
            String fromStructure, String toStructure,
            String action,
            Integer snapshotVersion, Long snapshotId,
            String comments, String performedBy) {
        return log(canevasId, fromStructure, toStructure, action,
                   snapshotVersion, snapshotId, comments, performedBy, null);
    }

    // ── Tous les logs — DC (pas de filtre) ───────────────────────────────────
    public List<CanevasWorkflowLog> getHistory(Long canevasId) {
        return workflowRepo.findByCanevasIdOrderByPerformedAtDesc(canevasId);
    }

    // ── Logs filtrés selon le rôle ────────────────────────────────────────────
    // typeStructure "1"=AGC, "2"=DR, autres=DC
    public List<CanevasWorkflowLog> getHistoryFiltered(
            Long canevasId, String typeStructure, String codeStructure) {

        if ("1".equals(typeStructure) && codeStructure != null) {
            // AGC : uniquement ses propres logs
            return workflowRepo.findByCanevasIdAndFromStructureOrderByPerformedAtDesc(
                    canevasId, codeStructure);
        }
        if ("2".equals(typeStructure) && codeStructure != null) {
            // DR : ses logs + logs des agences filles
            List<CanevasWorkflowLog> propres =
                workflowRepo.findByCanevasIdAndFromStructureOrderByPerformedAtDesc(
                        canevasId, codeStructure);
            List<CanevasWorkflowLog> agences =
                workflowRepo.findByCanevasIdAndStructureMereOrderByPerformedAtDesc(
                        canevasId, codeStructure);
            List<CanevasWorkflowLog> merged = new ArrayList<>(propres);
            merged.addAll(agences);
            merged.sort(Comparator.comparing(CanevasWorkflowLog::getPerformedAt).reversed());
            return merged;
        }
        // DC ou non précisé : tout
        return workflowRepo.findByCanevasIdOrderByPerformedAtDesc(canevasId);
    }
}

package net.kachout.saeb.web;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import net.kachout.saeb.dtos_.*;
import net.kachout.saeb.entities.CanevasStructureSnapshot;
import net.kachout.saeb.entities.CanevasWorkflowLog;
import net.kachout.saeb.service.CanevasSnapshotService;
import net.kachout.saeb.service.WorkflowLogService;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import java.util.List;

@RestController
@RequestMapping("/api/canevas")
@CrossOrigin
@RequiredArgsConstructor
@Slf4j
public class CanevasSnapshotController {

    private final CanevasSnapshotService snapshotService;
    private final WorkflowLogService workflowLogService;

    @PostMapping("/{id}/valider")
    public ResponseEntity<SnapshotResponseDto> valider(
            @PathVariable Long id,
            @RequestBody ValidationRequestDto request) {

        CanevasStructureSnapshot snapshot = snapshotService.createSnapshot(
                id, request.fromStructureCode(), request.fromStructureLabel(),
                request.matricule(), request.matricule(),
                request.observations(), "VALID_DONNEE", null);

        workflowLogService.log(id,
                request.fromStructureCode(), request.toStructureCode(),
                "VALIDER", snapshot.getVersionNumber(), snapshot.getSnapshotId(),
                request.observations(), request.matricule());

        return ResponseEntity.ok(new SnapshotResponseDto(
                snapshot.getSnapshotId(), snapshot.getVersionNumber(),
                request.fromStructureCode()));
    }

    @PostMapping("/{id}/refouler")
    public ResponseEntity<SnapshotResponseDto> refouler(
            @PathVariable Long id,
            @RequestBody RefoulementRequestDto request) {

        CanevasStructureSnapshot snapshot = snapshotService.createSnapshot(
                id, request.fromStructureCode(), request.fromStructureLabel(),
                request.matricule(), request.matricule(),
                request.motif(), "CORRIGE_DONNEE", null);

        workflowLogService.log(id,
                request.fromStructureCode(), request.toStructureCode(),
                "REFOULER", snapshot.getVersionNumber(), snapshot.getSnapshotId(),
                request.motif(), request.matricule());

        return ResponseEntity.ok(new SnapshotResponseDto(
                snapshot.getSnapshotId(), snapshot.getVersionNumber(),
                request.fromStructureCode()));
    }

    @GetMapping("/{id}/snapshots")
    public ResponseEntity<List<SnapshotSummaryDto>> getHistory(@PathVariable Long id) {
        return ResponseEntity.ok(snapshotService.getHistory(id));
    }

    @GetMapping("/{id}/snapshots/structure/{structureCode}")
    public ResponseEntity<SnapshotDetailDto> getByStructure(
            @PathVariable Long id, @PathVariable String structureCode) {
        return snapshotService.getByStructure(id, structureCode)
                .map(ResponseEntity::ok)
                .orElse(ResponseEntity.notFound().build());
    }

    // NOUVEAU — récupérer un snapshot par son ID exact (pour l'icône œil)
    @GetMapping("/{id}/snapshots/{snapshotId}/detail")
    public ResponseEntity<SnapshotDetailDto> getSnapshotById(
            @PathVariable Long id, @PathVariable Long snapshotId) {
        return snapshotService.getBySnapshotId(snapshotId)
                .map(ResponseEntity::ok)
                .orElse(ResponseEntity.notFound().build());
    }

    // MODIFIÉ — filtre par rôle via typeStructure + codeStructure
    @GetMapping("/{id}/snapshots/workflow-log")
    public ResponseEntity<List<CanevasWorkflowLog>> getWorkflowLog(
            @PathVariable Long id,
            @RequestParam(required = false) String typeStructure,
            @RequestParam(required = false) String codeStructure) {
        return ResponseEntity.ok(
                workflowLogService.getHistoryFiltered(id, typeStructure, codeStructure));
    }
}

package net.kachout.saeb.web;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;

    private final ObjectMapper objectMapper;

    public AvisController(AvisService avisService,
                          CanevasSnapshotService snapshotService,
                          WorkflowLogService workflowLogService,
                          ObjectMapper objectMapper) {
        this.avisService        = avisService;
        this.snapshotService    = snapshotService;
        this.workflowLogService = workflowLogService;
        this.objectMapper       = objectMapper;
    }

    @PostMapping
    public ResponseEntity<AvisDTO> saveAvis(@RequestBody AvisDTO avisDTO) {
        log.info("saveAvis → canevas={} statusAvis={}", avisDTO.getNumCaneva(), avisDTO.getStatusAvis());

        // 1. Sauvegarder l'avis EN PREMIER
        AvisDTO saved;
        try {
            saved = avisService.saveAvis(avisDTO);
        } catch (JsonProcessingException e) {
            log.error("Erreur sauvegarde avis", e);
            return ResponseEntity.badRequest().build();
        }

        // 2. Préparer les valeurs (plus de hardcoded)
        String structureCode  = String.valueOf(avisDTO.getStructure());
        String structureLabel = avisDTO.getStructureLabel() != null
                                ? avisDTO.getStructureLabel() : structureCode;
        String matricule      = avisDTO.getMatricule() != null
                                ? String.valueOf(avisDTO.getMatricule()) : "INCONNU";
        String observations   = avisDTO.getCommentaire() != null
                                ? avisDTO.getCommentaire() : "";
        String statusAtSnap   = resolveStatusAtSnapshot(avisDTO.getStatusAvis(), avisDTO.getAvisValue());

        // 3. Sérialiser l'avis en JSON pour le snapshot
        String avisJson = null;
        try {
            Map<String, Object> avisMap = new LinkedHashMap<>();
            avisMap.put("statusAvis",  avisDTO.getStatusAvis());
            avisMap.put("avisValue",   avisDTO.getAvisValue());
            avisMap.put("commentaire", avisDTO.getCommentaire());
            avisMap.put("structure",   avisDTO.getStructure());
            avisMap.put("matricule",   avisDTO.getMatricule());
            avisMap.put("avisId",      saved.getId());
            avisJson = objectMapper.writeValueAsString(avisMap);
        } catch (Exception e) {
            log.warn("Impossible de sérialiser l'avis", e);
        }

        // 4. Créer le snapshot APRÈS sauvegarde
        Long snapshotId  = null;
        Integer snapVer  = null;
        try {
            var snap = snapshotService.createSnapshot(
                    avisDTO.getNumCaneva(), structureCode, structureLabel,
                    matricule, matricule, observations, statusAtSnap, avisJson);
            snapshotId = snap.getSnapshotId();
            snapVer    = snap.getVersionNumber();
            log.info("✅ Snapshot v{} créé pour canevas={}", snapVer, avisDTO.getNumCaneva());
        } catch (Exception e) {
            log.warn("⚠️ Échec snapshot canevas={}", avisDTO.getNumCaneva(), e);
        }

        // 5. Créer le WorkflowLog
        try {
            String toStructure = avisDTO.getToStructureCode() != null
                                 ? avisDTO.getToStructureCode()
                                 : avisDTO.getCodeStructureReceptrice();
            String action = resolveWorkflowAction(avisDTO.getStatusAvis());
            // structureMere = structureMere de l'AvisDTO (valeur de l'agence vers sa DR)
            String structureMereLog = "1".equals(avisDTO.getTypeStructure())
                                     ? avisDTO.getStructureMere() : null;

            workflowLogService.log(avisDTO.getNumCaneva(),
                    structureCode, toStructure, action,
                    snapVer, snapshotId, observations, matricule,
                    structureMereLog);
            log.info("✅ WorkflowLog action={} canevas={}", action, avisDTO.getNumCaneva());
        } catch (Exception e) {
            log.warn("⚠️ Échec WorkflowLog canevas={}", avisDTO.getNumCaneva(), e);
        }

        return ResponseEntity.ok(saved);
    }

    // ── Méthodes privées ──────────────────────────────────────────────────────

    private String resolveStatusAtSnapshot(String statusAvis, String avisValue) {
        if (statusAvis == null) return "INCONNU";
        return switch (statusAvis) {
            case "PEC_CCA"        -> "PEC_DONNEE";
            case "PRES_VALIDER"   -> "VALID_DONNEE";
            case "CORRIGER"       -> "CORRIGE_DONNEE";
            case "PEC_AVIS_DEC"   -> "PEC_AVIS_DEC";
            case "CORRIGER2"      -> "CORRIGE_AVIS_DEC";
            case "VALIDER"        -> "VALIDER_DR";
            case "Envoyer"        -> "EN_COURS_ETUDE";
            case "Décision Finale"-> "1".equals(avisValue) ? "DEC_FINAL_PRISE_ACCORD" : "DEC_FINAL_PRISE_REJET";
            default               -> statusAvis;
        };
    }

    private String resolveWorkflowAction(String statusAvis) {
        if (statusAvis == null) return "ACTION_INCONNUE";
        return switch (statusAvis) {
            case "PEC_CCA"        -> "PRENDRE_EN_CHARGE";
            case "Envoyer"        -> "ENVOYER";
            case "PRES_VALIDER"   -> "PRE_VALIDER";
            case "CORRIGER"       -> "REFOULER";
            case "PEC_AVIS_DEC"   -> "PRENDRE_EN_CHARGE";
            case "VALIDER"        -> "VALIDER";
            case "CORRIGER2"      -> "REFOULER";
            case "Décision Finale"-> "DECISION_FINALE";
            default               -> statusAvis.toUpperCase();
        };
    }

import { Injectable } from '@angular/core';
import { HttpClient, HttpParams } from '@angular/common/http';
import { Observable } from 'rxjs';
import { environment } from '../../environments/environment.development';

export interface SnapshotSummary {
  snapshotId: number;
  structureCode: string;
  structureLabel: string;
  versionNumber: number;
  statusAtSnapshot: string;
  createdBy: string;
  matricule: string;
  createdAt: string;
  observations: string;
}

export interface EngagementPartie {
  typePartie: string; nom: string; cin?: string;
  numSeqPers: number; partCapital?: string;
  engagementsBNA: any[]; engagementsBCT: any[]; classeBNA: any[];
}

export interface EngagementDetaille {
  actionnaires: EngagementPartie[];
  premierResponsable: EngagementPartie | null;
  groupe: EngagementPartie[];
  vendeurs: EngagementPartie[];
}

export interface SnapshotDetail extends SnapshotSummary {
  engagementsBNA: any[]; engagementsBCT: any[];
  classeBNA: any[]; classeBCT: any[]; classeRetour: any[];
  engagementDetaille: EngagementDetaille;
  creditsData: any; exploitationData: any; garantiesData: any;
}

export interface ValidationRequest {
  fromStructureCode: string; fromStructureLabel: string;
  toStructureCode: string; matricule: string;
  observations: string; avisValue: 'FAVORABLE' | 'DEFAVORABLE' | 'SANS_AVIS';
}

export interface RefoulementRequest {
  fromStructureCode: string; fromStructureLabel: string;
  toStructureCode: string; matricule: string; motif: string;
}

export interface SnapshotResponse {
  snapshotId: number; versionNumber: number; structureCode: string;
}

export interface WorkflowLogEntry {
  logId: number; canevasId: number;
  fromStructure: string; toStructure: string;
  action: string; snapshotVersion: number; snapshotId: number;
  comments: string; performedBy: string; performedAt: string;
  structureMere?: string;
}

@Injectable({ providedIn: 'root' })
export class SnapshotService {

  private api = `${environment.apiURL}api/canevas`;

  constructor(private http: HttpClient) {}

  validerEtTransmettre(id: number, req: ValidationRequest): Observable<SnapshotResponse> {
    return this.http.post<SnapshotResponse>(`${this.api}/${id}/valider`, req);
  }

  refouler(id: number, req: RefoulementRequest): Observable<SnapshotResponse> {
    return this.http.post<SnapshotResponse>(`${this.api}/${id}/refouler`, req);
  }

  getHistory(id: number): Observable<SnapshotSummary[]> {
    return this.http.get<SnapshotSummary[]>(`${this.api}/${id}/snapshots`);
  }

  getByStructure(id: number, code: string): Observable<SnapshotDetail> {
    return this.http.get<SnapshotDetail>(`${this.api}/${id}/snapshots/structure/${code}`);
  }

  // NOUVEAU — récupérer un snapshot par son ID exact (pour l'icône œil)
  getBySnapshotId(canevasId: number, snapshotId: number): Observable<SnapshotDetail> {
    return this.http.get<SnapshotDetail>(
      `${this.api}/${canevasId}/snapshots/${snapshotId}/detail`
    );
  }

  // MODIFIÉ — passe typeStructure + codeStructure pour filtrage serveur
  getWorkflowLog(
    id: number,
    typeStructure?: string,
    codeStructure?: string
  ): Observable<WorkflowLogEntry[]> {
    let params = new HttpParams();
    if (typeStructure) params = params.set('typeStructure', typeStructure);
    if (codeStructure) params = params.set('codeStructure', codeStructure);
    return this.http.get<WorkflowLogEntry[]>(
      `${this.api}/${id}/snapshots/workflow-log`, { params }
    );
  }
}

import { Component, OnInit, signal, computed, inject } from '@angular/core';
import { MessageService } from 'primeng/api';
import {
  SnapshotService, SnapshotSummary, SnapshotDetail,
  SnapshotResponse, WorkflowLogEntry
} from '../../../../_services/snapshot.service';
import { CanevasAgrService } from '../../../../_services/canevas.agr.service';
import { UserService } from '../../../../_services/user.service';

@Component({
  selector: 'app-snapshot-history',
  templateUrl: './snapshot-history.component.html',
  styleUrls: ['./snapshot-history.component.css'],
  providers: [MessageService]
})
export class SnapshotHistoryComponent implements OnInit {

  private snapshotService = inject(SnapshotService);
  private canevasAgrService = inject(CanevasAgrService);
  private messageService = inject(MessageService);
  userService = inject(UserService);

  // ── State ────────────────────────────────────────────────────────
  history       = signal<SnapshotSummary[]>([]);
  workflowLog   = signal<WorkflowLogEntry[]>([]);
  selectedSnap  = signal<SnapshotDetail | null>(null);
  loading       = signal(false);
  activeTab     = signal<'log' | 'snapshots'>('log');
  showValidDlg  = signal(false);
  showRefoulDlg = signal(false);

  // ── Rôle utilisateur ─────────────────────────────────────────────
  private userType = computed(() => this.userService.user()?.typeStructure ?? '');
  private userCode = computed(() => this.userService.user()?.codeStructure ?? '');

  // ── Filtrage défensif côté client (le serveur filtre déjà) ───────
  filteredLogs = computed(() => {
    const logs = this.workflowLog();
    const type = this.userType();
    const code = this.userCode();
    if (type === '1' && code)
      return logs.filter(l => l.fromStructure === code || l.toStructure === code);
    if (type === '2' && code)
      return logs.filter(l => l.fromStructure === code || l.toStructure === code || l.structureMere === code);
    return logs;
  });

  filteredHistory = computed(() => this.history());
  totalVersions   = computed(() => this.history().length);
  totalLogs       = computed(() => this.filteredLogs().length);

  // Les boutons Valider/Refouler sont visibles uniquement pour DR(2) et DC(5)
  canAct = computed(() => this.userType() === '2' || this.userType() === '5');

  // ── Formulaires ───────────────────────────────────────────────────
  validForm = {
    fromStructureCode: '', fromStructureLabel: '',
    toStructureCode: '', matricule: '',
    observations: '', avisValue: 'FAVORABLE' as const
  };
  refoulForm = {
    fromStructureCode: '', fromStructureLabel: '',
    toStructureCode: '', matricule: '', motif: ''
  };
  avisOptions = [
    { label: 'Favorable',   value: 'FAVORABLE'  },
    { label: 'Défavorable', value: 'DEFAVORABLE' },
    { label: 'Sans avis',   value: 'SANS_AVIS'   }
  ];

  // ── Init ─────────────────────────────────────────────────────────
  ngOnInit(): void {
    const id = this.canevasAgrService.currentIdCaneva();
    if (id) this.loadAll(id);
    const u = this.userService.user();
    this.validForm.fromStructureCode  = u?.codeStructure ?? '';
    this.validForm.fromStructureLabel = u?.libStructure  ?? '';
    this.validForm.matricule          = u?.matricule     ?? '';
    this.refoulForm.fromStructureCode  = u?.codeStructure ?? '';
    this.refoulForm.fromStructureLabel = u?.libStructure  ?? '';
    this.refoulForm.matricule          = u?.matricule     ?? '';
  }

  private loadAll(id: number): void {
    this.loading.set(true);
    const u = this.userService.user();
    this.snapshotService.getHistory(id).subscribe({
      next: d => { this.history.set(d); this.loading.set(false); },
      error: () => this.loading.set(false)
    });
    this.snapshotService.getWorkflowLog(id, u?.typeStructure, u?.codeStructure).subscribe({
      next: d => this.workflowLog.set(d),
      error: e => console.error('WorkflowLog error', e)
    });
  }

  openSnapshot(snap: SnapshotSummary): void {
    const id = this.canevasAgrService.currentIdCaneva();
    if (!id) return;
    this.loading.set(true);
    this.snapshotService.getBySnapshotId(id, snap.snapshotId).subscribe({
      next:  d => { this.selectedSnap.set(d); this.loading.set(false); },
      error: () => {
        this.loading.set(false);
        this.messageService.add({ severity: 'error', summary: 'Erreur', detail: 'Snapshot introuvable' });
      }
    });
  }

  closeSnapshot(): void { this.selectedSnap.set(null); }

  valider(): void {
    const id = this.canevasAgrService.currentIdCaneva();
    if (!id) return;
    this.snapshotService.validerEtTransmettre(id, this.validForm).subscribe({
      next: (res: SnapshotResponse) => {
        this.messageService.add({ severity: 'success', summary: 'Validé',
          detail: `Snapshot v${res.versionNumber} créé` });
        this.showValidDlg.set(false);
        this.loadAll(id);
      },
      error: () => this.messageService.add({ severity: 'error', summary: 'Erreur', detail: 'Échec validation' })
    });
  }

  refouler(): void {
    const id = this.canevasAgrService.currentIdCaneva();
    if (!id || !this.refoulForm.motif?.trim()) {
      this.messageService.add({ severity: 'warn', summary: 'Attention', detail: 'Motif obligatoire' });
      return;
    }
    this.snapshotService.refouler(id, this.refoulForm).subscribe({
      next: (res: SnapshotResponse) => {
        this.messageService.add({ severity: 'warn', summary: 'Refoulé',
          detail: `Snapshot v${res.versionNumber}` });
        this.showRefoulDlg.set(false);
        this.loadAll(id);
      },
      error: () => this.messageService.add({ severity: 'error', summary: 'Erreur', detail: 'Échec refoulement' })
    });
  }

  // ── Helpers d'affichage ───────────────────────────────────────────
  actionSeverity(a: string): string {
    const m: Record<string, string> = {
      VALIDER:'success', PRE_VALIDER:'info', REFOULER:'danger',
      CORRIGER:'warning', ANNULER:'danger', PRENDRE_EN_CHARGE:'secondary',
      ENVOYER:'info', ENVOYER_SUPERVISEUR:'info', DECISION_FINALE:'success'
    };
    return m[a] ?? 'secondary';
  }

  statusSeverity(s: string): string {
    const m: Record<string, string> = {
      VALID_DONNEE:'success', CORRIGE_DONNEE:'danger',
      PEC_DONNEE:'info', PEC_AVIS_DEC:'warning',
      VALIDER_DR:'success', CORRIGE_AVIS_DEC:'danger',
      DEC_FINAL_PRISE_ACCORD:'success', DEC_FINAL_PRISE_REJET:'danger'
    };
    return m[s] ?? 'secondary';
  }

  actionLabel(a: string): string {
    const m: Record<string, string> = {
      VALIDER:'Valider', PRE_VALIDER:'Pré-Valider', REFOULER:'Refouler',
      CORRIGER:'Corriger', ANNULER:'Annuler',
      PRENDRE_EN_CHARGE:'Prise en charge', ENVOYER:'Envoyer',
      ENVOYER_SUPERVISEUR:'→ Superviseur', DECISION_FINALE:'Décision Finale'
    };
    return m[a] ?? a;
  }

  iconForAction(a: string): string {
    if (['VALIDER','PRE_VALIDER'].includes(a)) return 'pi pi-check';
    if (['REFOULER','CORRIGER'].includes(a))   return 'pi pi-times';
    if (a === 'DECISION_FINALE')               return 'pi pi-star';
    if (a === 'PRENDRE_EN_CHARGE')             return 'pi pi-inbox';
    return 'pi pi-send';
  }
}
