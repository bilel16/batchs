import org.springframework.stereotype.Service; import java.util.*; import java.util.stream.Collectors; @Service @RequiredArgsConstructor @Slf4j public class WorkflowLogService { private final CanevasWorkflowLogRepository repo; /** * Crée un log de transition workflow. * @param structureMere la DR parente de fromStructure (null si fromStructure est une DR ou DC) */ public CanevasWorkflowLog log( Long canevasId, String fromStructure, String toStructure, String action, Integer snapshotVersion, Long snapshotId, String comments, String performedBy, String structureMere // ✅ nouveau paramètre ) { CanevasWorkflowLog entry = CanevasWorkflowLog.builder() .canevasId(canevasId) .fromStructure(fromStructure) .toStructure(toStructure) .action(action) .snapshotVersion(snapshotVersion) .snapshotId(snapshotId) .comments(comments) .performedBy(performedBy) .structureMere(structureMere) // ✅ .build();
