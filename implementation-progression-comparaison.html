  // Helper statique — utilisé dans le component pour le diff
  static computeCreditsTotal(creditsData: any): { ct: number; inv: number; terrain: number; total: number } {
    const ct      = (creditsData?.courtTerme      ?? []).reduce((s: number, c: any) => s + (Number(c.montant) || 0), 0);
    const inv     = (creditsData?.investissement  ?? []).reduce((s: number, c: any) => s + (Number(c.montant) || 0), 0);
    const terrain = (creditsData?.achatTerrain    ?? []).reduce((s: number, c: any) => s + (Number(c.montant) || 0), 0);
    return { ct, inv, terrain, total: ct + inv + terrain };
  }

  // ── Calculer le total engagements BNA d'un snapshot ───────────────────────
  static computeEngagementsTotal(engagements: any[]): { count: number; totalEnc: number } {
    if (!Array.isArray(engagements)) return { count: 0, totalEnc: 0 };
    const totalEnc = engagements.reduce((s, e) => s + (Number(e.encours) || 0), 0);
    return { count: engagements.length, totalEnc };
  }

  // ── Nombre de sites dans exploitationData ─────────────────────────────────
  static computeSitesCount(exploitationData: any): number {
    return (exploitationData?.sites ?? []).length;
  }

import { Component, OnInit, signal, computed, inject } from '@angular/core';
import { MessageService } from 'primeng/api';
import {
  SnapshotService, SnapshotSummary, SnapshotDetail,
  SnapshotResponse, WorkflowLogEntry
} from '../../../../_services/snapshot.service';
import { CanevasAgrService } from '../../../../_services/canevas.agr.service';
import { UserService } from '../../../../_services/user.service';

// ── Interface pour une ligne de diff ──────────────────────────────────────────
interface DiffRow {
  label: string;
  snapValue: string;
  currentValue: string;
  changed: boolean;
  unit?: string;
}

// ── Étape de progression ──────────────────────────────────────────────────────
interface ProgressStep {
  label: string;
  sublabel: string;
  icon: string;
  done: boolean;
  active: boolean;
}

@Component({
  selector: 'app-snapshot-history',
  templateUrl: './snapshot-history.component.html',
  styleUrls: ['./snapshot-history.component.css'],
  providers: [MessageService]
})
export class SnapshotHistoryComponent implements OnInit {

  private snapshotService = inject(SnapshotService);
  private canevasAgrService = inject(CanevasAgrService);
  private messageService = inject(MessageService);
  userService = inject(UserService);

  // ── State principal ───────────────────────────────────────────────────────
  history       = signal<SnapshotSummary[]>([]);
  workflowLog   = signal<WorkflowLogEntry[]>([]);
  selectedSnap  = signal<SnapshotDetail | null>(null);
  loading       = signal(false);
  activeTab     = signal<'log' | 'snapshots'>('log');
  showValidDlg  = signal(false);
  showRefoulDlg = signal(false);

  // ── State pour la COMPARAISON ─────────────────────────────────────────────
  compareMode      = signal(false);                    // panneau diff visible
  compareOldSnap   = signal<SnapshotDetail | null>(null);  // snapshot historique (choisi)
  compareNewSnap   = signal<SnapshotDetail | null>(null);  // dernier snapshot = "état actuel"
  compareLoading   = signal(false);
  compareOldLabel  = signal('');   // "Snapshot v3 — 20/03/2024"
  compareNewLabel  = signal('');   // "État actuel — v4 — 22/03/2024"

  // ── Rôle utilisateur ─────────────────────────────────────────────────────
  private userType = computed(() => this.userService.user()?.typeStructure ?? '');
  private userCode = computed(() => this.userService.user()?.codeStructure ?? '');

  // ── Filtrage logs par rôle ────────────────────────────────────────────────
  filteredLogs = computed(() => {
    const logs = this.workflowLog();
    const type = this.userType();
    const code = this.userCode();
    if (type === '1' && code)
      return logs.filter(l => l.fromStructure === code || l.toStructure === code);
    if (type === '2' && code)
      return logs.filter(l => l.fromStructure === code || l.toStructure === code || l.structureMere === code);
    return logs;
  });

  filteredHistory = computed(() => this.history());
  totalVersions   = computed(() => this.history().length);
  totalLogs       = computed(() => this.filteredLogs().length);
  canAct          = computed(() => this.userType() === '2' || this.userType() === '5');

  // ── PROGRESSION DU DOSSIER ────────────────────────────────────────────────
  // Calculée à partir des actions dans le workflowLog
  progressSteps = computed((): ProgressStep[] => {
    const logs   = this.workflowLog();
    const snaps  = this.history();
    const actions = new Set(logs.map(l => l.action));
    const statuses = new Set(snaps.map(s => s.statusAtSnapshot));

    const has = (a: string) => actions.has(a);
    const hasStatus = (s: string) => statuses.has(s);

    // Étape 1 : Saisie Agence → toujours fait si des snapshots existent
    const step1done = snaps.length > 0;

    // Étape 2 : Validation Agence → PRES_VALIDER ou PRE_VALIDER action
    const step2done = has('PRE_VALIDER') || hasStatus('VALID_DONNEE');

    // Étape 3 : Analyse Direction Régionale → PEC_AVIS_DEC ou PRENDRE_EN_CHARGE en DR
    const step3done = has('VALIDER') || hasStatus('VALIDER_DR') || hasStatus('PEC_AVIS_DEC');

    // Étape 4 : Décision Direction Centrale → DECISION_FINALE
    const step4done = has('DECISION_FINALE') || hasStatus('DEC_FINAL_PRISE_ACCORD') || hasStatus('DEC_FINAL_PRISE_REJET');

    // Étape active = la première non-done
    const activeStep = !step1done ? 1 : !step2done ? 2 : !step3done ? 3 : !step4done ? 4 : 5;

    return [
      {
        label: 'Saisie',
        sublabel: 'Agence',
        icon: 'pi pi-pencil',
        done: step1done,
        active: activeStep === 1
      },
      {
        label: 'Validation',
        sublabel: 'Chef Agence',
        icon: 'pi pi-check-circle',
        done: step2done,
        active: activeStep === 2
      },
      {
        label: 'Analyse',
        sublabel: 'Dir. Régionale',
        icon: 'pi pi-search',
        done: step3done,
        active: activeStep === 3
      },
      {
        label: 'Décision',
        sublabel: 'Dir. Centrale',
        icon: 'pi pi-star',
        done: step4done,
        active: activeStep === 4
      }
    ];
  });

  currentProgressLabel = computed((): string => {
    const steps = this.progressSteps();
    const active = steps.find(s => s.active);
    if (active) return `En cours — ${active.label} (${active.sublabel})`;
    const done  = steps.filter(s => s.done);
    if (done.length === steps.length) return 'Dossier traité — Décision finale rendue';
    return 'En attente';
  });

  progressSeverity = computed((): string => {
    const statuses = new Set(this.history().map(s => s.statusAtSnapshot));
    if (statuses.has('DEC_FINAL_PRISE_ACCORD')) return 'success';
    if (statuses.has('DEC_FINAL_PRISE_REJET'))  return 'danger';
    if (statuses.has('CORRIGE_DONNEE') || statuses.has('CORRIGE_AVIS_DEC')) return 'warning';
    return 'info';
  });

  // ── DIFF — lignes calculées pour la comparaison ───────────────────────────
  diffRows = computed((): DiffRow[] => {
    const oldSnap = this.compareOldSnap();
    const newSnap = this.compareNewSnap();
    if (!oldSnap || !newSnap) return [];

    const rows: DiffRow[] = [];

    // ── Crédits ─────────────────────────────────────────
    const oldCred = SnapshotService.computeCreditsTotal(oldSnap.creditsData);
    const newCred = SnapshotService.computeCreditsTotal(newSnap.creditsData);

    rows.push({
      label: 'Crédit CT total',
      snapValue: this.fmt(oldCred.ct),
      currentValue: this.fmt(newCred.ct),
      changed: oldCred.ct !== newCred.ct,
      unit: 'TND'
    });
    rows.push({
      label: 'Crédit Investissement total',
      snapValue: this.fmt(oldCred.inv),
      currentValue: this.fmt(newCred.inv),
      changed: oldCred.inv !== newCred.inv,
      unit: 'TND'
    });
    rows.push({
      label: 'Crédit Achat Terrain total',
      snapValue: this.fmt(oldCred.terrain),
      currentValue: this.fmt(newCred.terrain),
      changed: oldCred.terrain !== newCred.terrain,
      unit: 'TND'
    });
    rows.push({
      label: 'Total crédits sollicités',
      snapValue: this.fmt(oldCred.total),
      currentValue: this.fmt(newCred.total),
      changed: oldCred.total !== newCred.total,
      unit: 'TND'
    });

    // ── Engagements BNA ─────────────────────────────────
    const oldBNA = SnapshotService.computeEngagementsTotal(oldSnap.engagementsBNA as any[]);
    const newBNA = SnapshotService.computeEngagementsTotal(newSnap.engagementsBNA as any[]);

    rows.push({
      label: 'Lignes engagements BNA',
      snapValue: String(oldBNA.count),
      currentValue: String(newBNA.count),
      changed: oldBNA.count !== newBNA.count,
      unit: 'lignes'
    });
    rows.push({
      label: 'Total encours BNA',
      snapValue: this.fmt(oldBNA.totalEnc),
      currentValue: this.fmt(newBNA.totalEnc),
      changed: oldBNA.totalEnc !== newBNA.totalEnc,
      unit: 'TND'
    });

    // ── Engagements BCT ─────────────────────────────────
    const oldBCT = SnapshotService.computeEngagementsTotal(oldSnap.engagementsBCT as any[]);
    const newBCT = SnapshotService.computeEngagementsTotal(newSnap.engagementsBCT as any[]);

    rows.push({
      label: 'Lignes engagements BCT',
      snapValue: String(oldBCT.count),
      currentValue: String(newBCT.count),
      changed: oldBCT.count !== newBCT.count,
      unit: 'lignes'
    });

    // ── Sites d'exploitation ─────────────────────────────
    const oldSites = SnapshotService.computeSitesCount(oldSnap.exploitationData);
    const newSites = SnapshotService.computeSitesCount(newSnap.exploitationData);

    rows.push({
      label: 'Sites d\'exploitation',
      snapValue: String(oldSites),
      currentValue: String(newSites),
      changed: oldSites !== newSites,
      unit: 'sites'
    });

    // ── Statut ──────────────────────────────────────────
    rows.push({
      label: 'Statut canvas',
      snapValue: oldSnap.statusAtSnapshot ?? '—',
      currentValue: newSnap.statusAtSnapshot ?? '—',
      changed: oldSnap.statusAtSnapshot !== newSnap.statusAtSnapshot
    });

    return rows;
  });

  // Nombre de différences
  diffCount = computed(() => this.diffRows().filter(r => r.changed).length);

  // ── Formulaires Valider / Refouler ────────────────────────────────────────
  validForm = {
    fromStructureCode: '', fromStructureLabel: '',
    toStructureCode: '', matricule: '',
    observations: '', avisValue: 'FAVORABLE' as const
  };
  refoulForm = {
    fromStructureCode: '', fromStructureLabel: '',
    toStructureCode: '', matricule: '', motif: ''
  };
  avisOptions = [
    { label: 'Favorable',   value: 'FAVORABLE'  },
    { label: 'Défavorable', value: 'DEFAVORABLE' },
    { label: 'Sans avis',   value: 'SANS_AVIS'   }
  ];

  // ── Init ──────────────────────────────────────────────────────────────────
  ngOnInit(): void {
    const id = this.canevasAgrService.currentIdCaneva();
    if (id) this.loadAll(id);
    const u = this.userService.user();
    this.validForm.fromStructureCode  = u?.codeStructure ?? '';
    this.validForm.fromStructureLabel = u?.libStructure  ?? '';
    this.validForm.matricule          = u?.matricule     ?? '';
    this.refoulForm.fromStructureCode  = u?.codeStructure ?? '';
    this.refoulForm.fromStructureLabel = u?.libStructure  ?? '';
    this.refoulForm.matricule          = u?.matricule     ?? '';
  }

  private loadAll(id: number): void {
    this.loading.set(true);
    const u = this.userService.user();
    this.snapshotService.getHistory(id).subscribe({
      next: d => { this.history.set(d); this.loading.set(false); },
      error: () => this.loading.set(false)
    });
    this.snapshotService.getWorkflowLog(id, u?.typeStructure, u?.codeStructure).subscribe({
      next: d => this.workflowLog.set(d),
      error: e => console.error('WorkflowLog error', e)
    });
  }

  // ── Actions : Voir (œil) ──────────────────────────────────────────────────
  openSnapshot(snap: SnapshotSummary): void {
    const id = this.canevasAgrService.currentIdCaneva();
    if (!id) return;
    this.compareMode.set(false);  // fermer le mode comparaison
    this.loading.set(true);
    this.snapshotService.getBySnapshotId(id, snap.snapshotId).subscribe({
      next:  d => { this.selectedSnap.set(d); this.loading.set(false); },
      error: () => {
        this.loading.set(false);
        this.messageService.add({ severity: 'error', summary: 'Erreur', detail: 'Snapshot introuvable' });
      }
    });
  }

  closeSnapshot(): void {
    this.selectedSnap.set(null);
    this.compareMode.set(false);
    this.compareOldSnap.set(null);
    this.compareNewSnap.set(null);
  }

  // ── Actions : Comparer (⚖) ────────────────────────────────────────────────
  openCompare(snap: SnapshotSummary): void {
    const id = this.canevasAgrService.currentIdCaneva();
    if (!id) return;

    // Trouver le dernier snapshot (état actuel)
    const allSnaps  = this.history();
    const latestSum = allSnaps.reduce((a, b) => a.versionNumber > b.versionNumber ? a : b, allSnaps[0]);

    if (!latestSum || latestSum.snapshotId === snap.snapshotId) {
      this.messageService.add({
        severity: 'warn',
        summary: 'Comparaison impossible',
        detail: 'Ce snapshot est déjà la version la plus récente.'
      });
      return;
    }

    this.compareLoading.set(true);
    this.compareMode.set(true);
    this.selectedSnap.set(null);
    this.compareOldSnap.set(null);
    this.compareNewSnap.set(null);

    // Charger le snapshot historique (ancien)
    this.snapshotService.getBySnapshotId(id, snap.snapshotId).subscribe({
      next: old => {
        this.compareOldSnap.set(old);
        this.compareOldLabel.set(`Snapshot v${snap.versionNumber} — ${this.fmtDate(snap.createdAt)}`);

        // Charger le dernier snapshot (actuel)
        this.snapshotService.getBySnapshotId(id, latestSum.snapshotId).subscribe({
          next: cur => {
            this.compareNewSnap.set(cur);
            this.compareNewLabel.set(`État actuel v${latestSum.versionNumber} — ${this.fmtDate(latestSum.createdAt)}`);
            this.compareLoading.set(false);
          },
          error: () => this.compareLoading.set(false)
        });
      },
      error: () => this.compareLoading.set(false)
    });
  }

  closeCompare(): void {
    this.compareMode.set(false);
    this.compareOldSnap.set(null);
    this.compareNewSnap.set(null);
  }

  // ── Actions Valider / Refouler ────────────────────────────────────────────
  valider(): void {
    const id = this.canevasAgrService.currentIdCaneva();
    if (!id) return;
    this.snapshotService.validerEtTransmettre(id, this.validForm).subscribe({
      next: (res: SnapshotResponse) => {
        this.messageService.add({ severity: 'success', summary: 'Validé',
          detail: `Snapshot v${res.versionNumber} créé` });
        this.showValidDlg.set(false);
        this.loadAll(id);
      },
      error: () => this.messageService.add({ severity: 'error', summary: 'Erreur', detail: 'Échec validation' })
    });
  }

  refouler(): void {
    const id = this.canevasAgrService.currentIdCaneva();
    if (!id || !this.refoulForm.motif?.trim()) {
      this.messageService.add({ severity: 'warn', summary: 'Attention', detail: 'Motif obligatoire' });
      return;
    }
    this.snapshotService.refouler(id, this.refoulForm).subscribe({
      next: (res: SnapshotResponse) => {
        this.messageService.add({ severity: 'warn', summary: 'Refoulé',
          detail: `Snapshot v${res.versionNumber}` });
        this.showRefoulDlg.set(false);
        this.loadAll(id);
      },
      error: () => this.messageService.add({ severity: 'error', summary: 'Erreur', detail: 'Échec refoulement' })
    });
  }

  // ── Helpers visuels ───────────────────────────────────────────────────────
  actionSeverity(a: string): string {
    const m: Record<string, string> = {
      VALIDER:'success', PRE_VALIDER:'info', REFOULER:'danger',
      CORRIGER:'warning', ANNULER:'danger', PRENDRE_EN_CHARGE:'secondary',
      ENVOYER:'info', ENVOYER_SUPERVISEUR:'info', DECISION_FINALE:'success'
    };
    return m[a] ?? 'secondary';
  }

  statusSeverity(s: string): string {
    const m: Record<string, string> = {
      VALID_DONNEE:'success', CORRIGE_DONNEE:'danger',
      PEC_DONNEE:'info', PEC_AVIS_DEC:'warning',
      VALIDER_DR:'success', CORRIGE_AVIS_DEC:'danger',
      DEC_FINAL_PRISE_ACCORD:'success', DEC_FINAL_PRISE_REJET:'danger'
    };
    return m[s] ?? 'secondary';
  }

  actionLabel(a: string): string {
    const m: Record<string, string> = {
      VALIDER:'Valider', PRE_VALIDER:'Pré-Valider', REFOULER:'Refouler',
      CORRIGER:'Corriger', ANNULER:'Annuler',
      PRENDRE_EN_CHARGE:'Prise en charge', ENVOYER:'Envoyer',
      ENVOYER_SUPERVISEUR:'→ Superviseur', DECISION_FINALE:'Décision Finale'
    };
    return m[a] ?? a;
  }

  iconForAction(a: string): string {
    if (['VALIDER','PRE_VALIDER'].includes(a)) return 'pi pi-check';
    if (['REFOULER','CORRIGER'].includes(a))   return 'pi pi-times';
    if (a === 'DECISION_FINALE')               return 'pi pi-star';
    if (a === 'PRENDRE_EN_CHARGE')             return 'pi pi-inbox';
    return 'pi pi-send';
  }

  // ── Formatage ─────────────────────────────────────────────────────────────
  private fmt(n: number): string {
    return new Intl.NumberFormat('fr-TN').format(n);
  }

  fmtDate(d: string): string {
    if (!d) return '—';
    return new Date(d).toLocaleDateString('fr-TN', { day:'2-digit', month:'2-digit', year:'numeric' });
  }

  diffArrow(row: DiffRow): string {
    if (!row.changed) return '=';
    const a = parseFloat(row.snapValue.replace(/\s/g, ''));
    const b = parseFloat(row.currentValue.replace(/\s/g, ''));
    if (isNaN(a) || isNaN(b)) return '≠';
    return b > a ? '↑' : '↓';
  }

  diffClass(row: DiffRow): string {
    if (!row.changed) return 'diff-same';
    const a = parseFloat(row.snapValue.replace(/\s/g, ''));
    const b = parseFloat(row.currentValue.replace(/\s/g, ''));
    if (isNaN(a) || isNaN(b)) return 'diff-changed';
    return b > a ? 'diff-up' : 'diff-down';
  }
}
